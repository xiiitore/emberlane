import tempfile
import unittest
from pathlib import Path
from emberlane.meta_planner import MetaPlanner


class MetaPlannerTests(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.root = Path(self.tmp.name)
        self.planner = MetaPlanner(self.root)

    def tearDown(self):
        self.tmp.cleanup()

    def test_first_task_is_persisted(self):
        task = self.planner.next_task()
        state = self.planner._load()
        self.assertEqual(task["id"], "META-001")
        self.assertEqual(state["active"]["id"], "META-001")

    def test_next_resumes_active_task(self):
        first = self.planner.next_task()
        second = self.planner.next_task()
        self.assertEqual(first, second)

    def test_pass_advances_backlog(self):
        task = self.planner.next_task()
        self.planner.record_result(task["id"], "PASS",
                                   "Targeted test output verified.")
        next_task = self.planner.next_task()
        self.assertEqual(next_task["id"], "META-002")

    def test_kill_is_terminal_for_task(self):
        task = self.planner.next_task()
        self.planner.record_result(task["id"], "KILL",
                                   "Inspection proved this approach invalid.")
        self.assertEqual(self.planner.next_task()["id"], "META-002")

    def test_blocked_is_terminal_for_task(self):
        task = self.planner.next_task()
        self.planner.record_result(task["id"], "BLOCKED",
                                   "Required independent evidence unavailable.")
        self.assertEqual(self.planner.next_task()["id"], "META-002")

    def test_rejects_wrong_task_id(self):
        self.planner.next_task()
        with self.assertRaises(ValueError):
            self.planner.record_result("META-002", "PASS",
                                       "This result belongs to another task.")

    def test_rejects_insufficient_evidence(self):
        task = self.planner.next_task()
        with self.assertRaises(ValueError):
            self.planner.record_result(task["id"], "PASS", "ok")

    def test_backlog_exhaustion(self):
        for task_id, _, _ in __import__(
                "emberlane.meta_planner", fromlist=["TASKS"]).TASKS:
            task = self.planner.next_task()
            self.assertEqual(task["id"], task_id)
            self.planner.record_result(task_id, "PASS",
                                       "Test evidence recorded for task.")
        self.assertIsNone(self.planner.next_task())


if __name__ == "__main__":
    unittest.main()
