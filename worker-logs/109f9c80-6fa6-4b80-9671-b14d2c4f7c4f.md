import importlib.util
import json
import tempfile
import unittest
from pathlib import Path
from unittest.mock import patch

from emberlane.meta_planner import MetaPlanner


SCRIPT = Path(__file__).resolve().parents[1] / "scripts" / "emberlane-autonomous-loop.py"
SPEC = importlib.util.spec_from_file_location("emberlane_autonomous_loop", SCRIPT)
LOOP = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(LOOP)


class AutonomousMetaAdapterTests(unittest.TestCase):
    def test_experimental_id_does_not_call_meta_planner(self):
        with tempfile.TemporaryDirectory() as tmp:
            root = Path(tmp)
            log = root / "worker-logs" / "autonomous-planner.log"
            log.parent.mkdir(parents=True)

            with patch.object(LOOP, "ROOT", root), patch.object(LOOP, "LOG", log):
                result = LOOP.record_meta_result(
                    {"meta_id": "EXP-QUEUE-01"},
                    "KILL",
                    "Executor result failed strict validation.",
                )

            self.assertFalse(result)
            self.assertFalse((root / ".emberlane/meta-planner-state.json").exists())
            self.assertIn("NON_META_RESULT id=EXP-QUEUE-01", log.read_text())

    def test_real_meta_id_is_recorded(self):
        with tempfile.TemporaryDirectory() as tmp:
            root = Path(tmp)
            planner = MetaPlanner(root)
            task = planner.next_task(quiet=True)

            with patch.object(LOOP, "ROOT", root):
                result = LOOP.record_meta_result(
                    {"meta_id": task["id"]},
                    "KILL",
                    "Focused regression verification failed as recorded.",
                )

            self.assertTrue(result)
            state = json.loads(
                (root / ".emberlane/meta-planner-state.json").read_text()
            )
            self.assertIsNone(state["active"])
            self.assertEqual(state["history"][-1]["status"], "KILL")

    def test_stale_meta_result_does_not_mutate_active_task(self):
        with tempfile.TemporaryDirectory() as tmp:
            root = Path(tmp)
            planner = MetaPlanner(root)
            active = planner.next_task(quiet=True)
            log = root / "worker-logs" / "autonomous-planner.log"
            log.parent.mkdir(parents=True)

            with patch.object(LOOP, "ROOT", root), patch.object(LOOP, "LOG", log):
                result = LOOP.record_meta_result(
                    {"meta_id": "META-002"},
                    "KILL",
                    "A stale result must not mutate the current active meta task.",
                )

            self.assertFalse(result)
            state = planner._load()
            self.assertEqual(state["active"]["id"], active["id"])
            self.assertEqual(state["history"][-1]["status"], "IN_PROGRESS")
            self.assertIn("META_RESULT_SKIPPED", log.read_text())

    def test_unknown_meta_namespace_is_rejected(self):
        with tempfile.TemporaryDirectory() as tmp:
            root = Path(tmp)
            with patch.object(LOOP, "ROOT", root):
                with self.assertRaisesRegex(ValueError, "Unknown meta task id"):
                    LOOP.record_meta_result(
                        {"meta_id": "META-999"},
                        "KILL",
                        "Unknown meta-task identifiers must not be accepted.",
                    )
            self.assertFalse((root / ".emberlane/meta-planner-state.json").exists())


if __name__ == "__main__":
    unittest.main()
