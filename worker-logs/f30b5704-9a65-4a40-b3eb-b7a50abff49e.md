import tempfile
import unittest
from pathlib import Path

from emberlane import Workspace, WorkspaceError


class WorkspaceTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name) / "workspace"
        self.ws = Workspace(self.root, max_file_bytes=16)

    def test_max_file_bytes_must_be_a_positive_integer(self):
        for value in (0, -1, True, 1.5, "16"):
            with self.subTest(value=value), self.assertRaisesRegex(
                ValueError, "positive integer"
            ):
                Workspace(self.root / "invalid-limit", max_file_bytes=value)

    def test_write_read_and_list(self):
        self.ws.write_text("src/main.py", "print('ok')\n")
        self.assertEqual(self.ws.read_text("src/main.py"), "print('ok')\n")
        self.assertEqual(self.ws.list_files(), ["src/main.py"])

    def test_rejects_parent_traversal(self):
        with self.assertRaises(WorkspaceError):
            self.ws.write_text("../escape.txt", "no")
        with self.assertRaises(WorkspaceError):
            self.ws.read_text("../escape.txt")

    def test_validate_write_path_rejects_nul_and_backslash(self):
        for path in ("bad\x00name", "bad\\name"):
            with self.subTest(path=path), self.assertRaises(WorkspaceError):
                self.ws.validate_write_path(path)

    def test_rejects_noncanonical_relative_paths(self):
        for path in ("src//file.py", "src/./file.py"):
            with self.subTest(path=path), self.assertRaises(WorkspaceError):
                self.ws.validate_write_path(path)
            with self.subTest(path=path), self.assertRaises(WorkspaceError):
                self.ws.read_text(path)
