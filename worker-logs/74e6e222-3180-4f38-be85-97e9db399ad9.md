import os
import subprocess
import tempfile
import unittest


ROOT = Path(__file__).resolve().parents[1]

class CodexAutopilotSupervisorTests(unittest.TestCase):

        result = subprocess.run(
            ["bash", "-n", str(SCRIPT)],
            text=True,
            capture_output=True,
            check=False,
        )
        self.assertEqual(result.returncode, 0, result.stderr)

    def test_usage_limit_is_terminal_before_generic_failure_counter(self):
        source = SCRIPT.read_text(encoding="utf-8")
        usage_branch = source.index('SUPERVISOR_RESULT=CODEX_USAGE_LIMIT')
        generic_failure_branch = source.index('record "SESSION_FAILURES=$consecutive_failures/3"')
        self.assertLess(usage_branch, generic_failure_branch)
        self.assertIn(
            'grep -Eiq "you.ve hit your usage limit|usage limit|rate limit|too many requests|quota exceeded" "$child_log"',
            source,
        )
        self.assertIn('CODEX_RETRY=SUPPRESSED_UNTIL_LIMIT_CLEARS', source)

            home = Path(tmp)
            project = home / "emberlane" / "emberlane-connected"
            (project / "scripts").mkdir(parents=True)
            (project / "worker-logs").mkdir()
            (project / ".emberlane").mkdir()
