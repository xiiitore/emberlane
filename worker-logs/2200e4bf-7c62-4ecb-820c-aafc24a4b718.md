"""Read-only deterministic planner for the Emberlane task queue."""
from __future__ import annotations

import argparse
import json
from collections import Counter
from pathlib import Path
from typing import Any

from .pause import is_paused
from .queue import TaskQueue


def build_plan(
    root: str | Path = ".",
    state_path: str | Path | None = None,
) -> dict[str, Any]:
    project_root = Path(root).resolve()
    queue_path = (
        Path(state_path).resolve()
        if state_path is not None
        else project_root / ".emberlane" / "queue.json"
    )

    if is_paused(project_root):
        paused = True
    else:
        paused = False

    if queue_path.exists() and not queue_path.is_file():
        raise ValueError(f"queue path exists but is not a regular file: {queue_path}")

    if state_path is not None and not queue_path.is_file():
        return {
            "planner": "emberlane",
            "mode": "read_only",
            "decision": "QUEUE_MISSING",
            "paused": paused,
            "queue_path": str(queue_path),
            "counts": {},
            "failed_tasks": [],
            "next_action": "inspect_queue_configuration",
            "execution_performed": False,
        }

    if queue_path.is_file():
        tasks = TaskQueue(queue_path).list_tasks()
    else:
        # TaskQueue does not persist a file until the first queue mutation.
        # A missing file therefore represents an empty queue at a fresh path.
        tasks = []
    task_results_need_review = [
        {
            "task_id": task.task_id,
            "kind": task.kind,
            "result_status": (
                task.result.get("status")
                if isinstance(task.result, dict)
                else None
            ),
            "applied": (
                task.result.get("applied")
                if isinstance(task.result, dict)
                else None
            ),
        }
        for task in tasks
        if task.status == "completed"
        and task.kind == "local_model_edit"
        and not (
            isinstance(task.result, dict)
            and task.result.get("status") == "applied_and_verified"
            and task.result.get("applied") is True
            and task.result.get("tests") == "passed"
        )
    ]
    counts = Counter(task.status for task in tasks)

    failed_tasks = [
        {
            "task_id": task.task_id,
            "kind": task.kind,
            "attempts": task.attempts,
            "has_error": bool(task.last_error),
        }
        for task in tasks
        if task.status == "failed"
    ]

    if paused:
        decision = "PAUSED"
        next_action = "operator_review_pause_before_execution"
    elif counts["running"]:
        decision = "RUNNING_TASKS_NEED_REVIEW"
        next_action = "inspect_running_tasks_before_dispatch"
    elif counts["failed"]:
        decision = "FAILED_TASKS_NEED_REVIEW"
        next_action = "inspect_failed_tasks_without_retrying_them"
    elif task_results_need_review:
        decision = "TASK_RESULTS_NEED_REVIEW"
        next_action = "inspect_completed_task_results_before_dispatch"
    elif counts["pending"]:
        decision = "PENDING_TASKS_EXIST"
        next_action = "review_pending_tasks_before_manual_execution"
    else:
        decision = "READY_FOR_EXPLICIT_TASK"
        next_action = "select_an_explicit_task_such_as_run_tests"

    return {
        "planner": "emberlane",
        "mode": "read_only",
        "decision": decision,
        "paused": paused,
        "queue_path": str(queue_path),
        "total_tasks": len(tasks),
        "counts": dict(sorted(counts.items())),
        "failed_tasks": failed_tasks,
        "task_results_need_review": task_results_need_review,
        "next_action": next_action,
        "execution_performed": False,
        "automatic_retries": False,
    }


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(
        prog="emberlane-planner",
        description="Inspect the queue and recommend the next safe action.",
    )
    parser.add_argument("--root", default=".", help="project root")
    parser.add_argument("--state", default=None, help="optional queue JSON path")
    args = parser.parse_args(argv)

    try:
        result = build_plan(args.root, args.state)
    except (OSError, ValueError) as exc:
        print(json.dumps({
            "planner": "emberlane",
            "decision": "INSPECTION_ERROR",
            "error_type": type(exc).__name__,
            "error": str(exc),
            "execution_performed": False,
        }, ensure_ascii=False, indent=2))
        return 2

    print(json.dumps(result, ensure_ascii=False, indent=2))
    return 0 if result["decision"] != "QUEUE_MISSING" else 2


if __name__ == "__main__":
    raise SystemExit(main())
