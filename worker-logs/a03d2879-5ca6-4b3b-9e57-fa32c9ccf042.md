def log(message):

def save(state):
    tmp = STATE.with_suffix(".tmp")
    tmp.write_text(json.dumps(state, ensure_ascii=False) + "\n", encoding="utf-8")
    os.replace(tmp, STATE)

def queue_tasks():
    if not QUEUE.exists():

        raise ValueError("queue.json is not a list")
    return data


def submit(kind, payload, idempotency_key=None):
    if kind not in ALLOWED or not isinstance(payload, dict):
        raise ValueError(f"task kind/payload rejected: {kind}")
    cmd = [
        sys.executable, "-m", "emberlane", "submit", kind,
        "--payload", json.dumps(payload, ensure_ascii=False),
    ]
    if idempotency_key:
        cmd.extend(["--idempotency-key", idempotency_key])
    result = subprocess.run(cmd, cwd=ROOT, capture_output=True, text=True)
    if result.returncode:
        raise RuntimeError(
            f"submit failed: {result.stderr[-1500:]} {result.stdout[-1500:]}"
        )
    data = json.loads(result.stdout)
    task_id = data.get("task_id")
    if not task_id:
        raise ValueError(f"submit output lacks task_id: {result.stdout[-1000:]}")
    return task_id
