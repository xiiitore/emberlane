def _make_patch(before: dict[str, bytes], after: dict[str, bytes], changed: list[str], limit: int) -> str:
    pieces: list[str] = []

        new = after.get(name, b"").decode("utf-8", errors="replace").splitlines(keepends=True)

                  f"+++ b/{name}\n" if name in after else "+++ /dev/null\n"]
